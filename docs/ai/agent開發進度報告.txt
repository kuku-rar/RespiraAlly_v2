  🏆 最終推薦方案

  CrewAI memory=False + Repository Pattern

  架構實現：

  # ==========================================
  # Domain Layer - 領域層（業務邏輯）
  # ==========================================
  from abc import ABC, abstractmethod
  from typing import List
  from dataclasses import dataclass

  @dataclass
  class Message:
      role: str
      content: str
      timestamp: datetime

  @dataclass
  class Document:
      content: str
      metadata: dict
      score: float

  class ConversationRepository(ABC):
      """對話歷史存儲接口"""
      @abstractmethod
      async def get_history(self, user_id: str, limit: int = 10) -> List[Message]:
          pass

      @abstractmethod
      async def save_message(self, user_id: str, message: Message) -> None:
          pass

      @abstractmethod
      async def clear_history(self, user_id: str) -> None:
          pass

  class KnowledgeRepository(ABC):
      """COPD 知識庫檢索接口"""
      @abstractmethod
      async def search(
          self, 
          query: str, 
          top_k: int = 3,
          category: str | None = None
      ) -> List[Document]:
          pass

  # ==========================================
  # Infrastructure Layer - 基礎設施層（實現）
  # ==========================================
  class RedisConversationRepository(ConversationRepository):
      def __init__(self, redis_client: Redis):
          self.redis = redis_client
          self.ttl = 300  # 5 minutes

      async def get_history(self, user_id: str, limit: int = 10) -> List[Message]:
          key = f"conv:{user_id}"
          messages_json = await self.redis.lrange(key, -limit, -1)
          return [Message(**json.loads(msg)) for msg in messages_json]

      async def save_message(self, user_id: str, message: Message) -> None:
          key = f"conv:{user_id}"
          await self.redis.rpush(key, message.json())
          await self.redis.expire(key, self.ttl)

  class PgvectorKnowledgeRepository(KnowledgeRepository):
      def __init__(self, db_session: AsyncSession, openai_client: OpenAI):
          self.db = db_session
          self.openai = openai_client

      async def search(
          self, 
          query: str, 
          top_k: int = 3,
          category: str | None = None
      ) -> List[Document]:
          # 1. 生成 embedding
          response = await self.openai.embeddings.create(
              model="text-embedding-3-small",
              input=query
          )
          query_embedding = response.data[0].embedding

          # 2. pgvector 相似度搜索
          stmt = select(COPDKnowledgeBaseModel).order_by(
              COPDKnowledgeBaseModel.embedding.cosine_distance(query_embedding)
          ).limit(top_k)

          if category:
              stmt = stmt.where(COPDKnowledgeBaseModel.category == category)

          result = await self.db.execute(stmt)
          rows = result.scalars().all()

          return [
              Document(
                  content=f"Q: {row.question}\nA: {row.answer}",
                  metadata={"category": row.category, "keywords": row.keywords},
                  score=1.0  # pgvector 會返回 distance，可轉換
              )
              for row in rows
          ]

  # ==========================================
  # Application Layer - 應用層（協調）
  # ==========================================
  from crewai import Crew, Agent, Task

  class AgentManager:
      def __init__(
          self,
          conversation_repo: ConversationRepository,
          knowledge_repo: KnowledgeRepository,
          openai_api_key: str
      ):
          self.conv_repo = conversation_repo
          self.kb_repo = knowledge_repo

          # CrewAI Agents (memory=False)
          self.guardrail_agent = Agent(
              role="Safety Checker",
              goal="檢查訊息是否包含醫療風險、法律問題或不當內容",
              backstory="嚴謹的安全審查器，保護用戶安全",
              llm=ChatOpenAI(model="gpt-4-turbo", api_key=openai_api_key),
              memory=False,  # ← 關鍵：不用 CrewAI Memory
              allow_delegation=False,
              verbose=True
          )

          self.health_agent = Agent(
              role="COPD Health Companion",
              goal="提供專業、溫暖的 COPD 健康建議",
              backstory="經驗豐富的呼吸科護理師，擅長 COPD 衛教",
              llm=ChatOpenAI(model="gpt-4-turbo", api_key=openai_api_key),
              memory=False,  # ← 關鍵：不用 CrewAI Memory
              allow_delegation=False,
              verbose=True
          )

          # Crew 協調
          self.crew = Crew(
              agents=[self.guardrail_agent, self.health_agent],
              memory=False,  # ← 關鍵：不用 CrewAI Memory
              verbose=True
          )

      async def process_message(self, user_id: str, message: str) -> str:
          # 1. 取得對話歷史 (from Redis)
          history = await self.conv_repo.get_history(user_id)
          history_text = "\n".join([
              f"{msg.role}: {msg.content}" for msg in history
          ])

          # 2. 檢索 COPD 知識 (from pgvector)
          context_docs = await self.kb_repo.search(message, top_k=3)
          context_text = "\n\n".join([
              f"[知識 {i+1}] {doc.content}" for i, doc in enumerate(context_docs)
          ])

          # 3. 定義 Tasks
          guardrail_task = Task(
              description=f"""
              檢查以下使用者訊息是否安全：
  
              訊息：{message}
  
              檢查項目：
              - 是否包含自殺/自殘念頭
              - 是否需要緊急醫療處理
              - 是否涉及法律問題
  
              如果安全，回傳 "SAFE"
              如果不安全，回傳 "UNSAFE: [原因]"
              """,
              agent=self.guardrail_agent,
              expected_output="SAFE 或 UNSAFE: [原因]"
          )

          health_task = Task(
              description=f"""
              根據以下資訊，生成專業且溫暖的回應：
  
              使用者訊息：{message}
  
              對話歷史：
              {history_text}
  
              COPD 知識庫：
              {context_text}
  
              要求：
              - 使用繁體中文
              - 語氣溫暖親切
              - 引用知識庫內容時註明來源
              - 若涉及醫療建議，提醒諮詢醫師
              - 控制在 150 字以內
              """,
              agent=self.health_agent,
              expected_output="一段溫暖且專業的回應",
              context=[guardrail_task]  # ← 依賴 guardrail 結果
          )

          # 4. 執行 Crew (Guardrail → Health)
          result = await self.crew.kickoff_async(
              tasks=[guardrail_task, health_task]
          )

          # 5. 檢查 Guardrail 結果
          if "UNSAFE" in guardrail_task.output.raw_output:
              response = "抱歉，您的訊息包含敏感內容。如需緊急協助，請撥打 119。"
          else:
              response = health_task.output.raw_output

          # 6. 保存對話 (to Redis)
          await self.conv_repo.save_message(user_id, Message(
              role="user", content=message, timestamp=datetime.utcnow()
          ))
          await self.conv_repo.save_message(user_id, Message(
              role="assistant", content=response, timestamp=datetime.utcnow()
          ))

          return response

  📋 實施決策建議

  ✅ 推薦：CrewAI memory=False

  理由總結：
  1. ✅ beloved_grandson 實證：已驗證可行
  2. ✅ DDD 完美契合：Repository Pattern + 依賴反轉
  3. ✅ 最簡架構：4 個核心概念，2 層調試
  4. ✅ 維護性最優：組件可獨立替換
  5. ✅ 部署最輕：~210MB vs ~560MB (Mem0) vs ~660MB (Chroma)
  6. ✅ Linus 認證："Bad programmers worry about the code. Good programmers worry about data structures." - 我們的數據結構清晰（Redis        
  對話, pgvector 知識）

  ❌ 不推薦：Mem0 / Custom Storage

  Mem0 的問題：
  - 引入第5層抽象（User → LINE → RabbitMQ → AgentManager → CrewAI → Mem0 → pgvector）
  - 破壞 DDD 封裝（Domain 依賴 Mem0 API）
  - beloved_grandson 沒用（失去參考價值）

  Custom Storage 的問題：
  - 實現成本高（需實現完整 Storage 接口）
  - 維護成本高（CrewAI 更新需跟進）
  - 測試成本高（需驗證自訂實現正確性）
